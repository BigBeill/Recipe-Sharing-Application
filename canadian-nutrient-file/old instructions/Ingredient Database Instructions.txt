// ============ START OF QUERY ============

CREATE TABLE food (
	_id numeric,
	code numeric,
   	food_group_id numeric,
	food_source_id numeric,
	description text,
	description_f text,
	date_of_entry date,
	date_of_publication date,
	country_code numeric,
	scientific_name text
);

CREATE TABLE nutrient_amount (
	food_id numeric,
   	nutrient_id numeric,
   	value numeric,
	standard_error numeric,
	number_of_observations numeric,
	nutrient_source_id numeric,
	date_entry date
);

CREATE TABLE conversion_factor (
	food_id numeric,
	measure_id numeric,
	value numeric,
	date_of_entry date
);

CREATE TABLE refuse_amount (
	food_id numeric,
	refuse_id numeric,
	refuse_amount numeric,
	date_of_entry date
);

CREATE TABLE yield_amount (
	food_id numeric,
	yield_id numeric,
	yield_amount numeric,
	date_of_entry date
);

CREATE TABLE food_group (
	food_group_id numeric,
	food_group_code numeric,
	description text,
	description_f text
);

CREATE TABLE food_source (
	_id numeric,
	code numeric,
	description text,
	description_f text
);

CREATE TABLE nutrient (
	_id numeric,
	code numeric,
	symbol text,
	unit text,
	description text,
	description_f text,
	tag_name text,
	decimals numeric
);

CREATE TABLE nutrient_source (
	_id numeric,
	code numeric,
	description text,
	description_f text
);

CREATE TABLE measure (
	_id numeric,
	description text,
	description_f text
);

CREATE TABLE refuse (
	_id numeric,
	description text,
	description_f text
);

CREATE TABLE yield (
	_id numeric,
	description text,
	description_f text
);

CREATE TABLE common_food_names (
	food_id numeric,
	common_name text
);

// ============ END OF QUERY ============
